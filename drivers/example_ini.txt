; Optional: pads without an ini get a default map from their HID descriptor,
; see docs/usb_gamepads.md. An ini is only needed to change that map.
;
[info]
;
; USB HID joystick VID/PID (hex values, lowercase)
;
vid = 081f
pid = e401
;
; Optional internal name of the device
;
name = Retrolink USB Gamepad
;
; Optional: use only the reports with this id (decimal). Without it every
; report of the device is mapped. Set it for pads that also send battery or
; vendor reports, so those do not release the buttons.
;
; report_id = 1

[buttons]
;
; Mapping to MD2 gamepad buttons:
; up, down, left, right, a, b, c, x, y, z, start, mode
;
; Each button is one or more sources separated by |, no spaces pls.
; A source is pos,type,value or pos,type,value,arg:
; - pos: the byte position in the report (decimal), starting from 0,
;   after the report id byte if the pad has report ids
; - type and value:
;   0 = the byte equals value (0..255)
;   1 = bit number value (0..7) is set
;   2 = hat switch: the low 4 bits of the byte, shifted right by arg
;       (0..4, default 0), equal value (0..7, 0 = up, clockwise) or a
;       diagonal next to it; 8..15 is centered
;   3 = value <= byte <= arg (an axis range)
;   4 = bit number value (0..7) is clear
;
up = 1,0,0
down = 1,0,255
left = 0,0,0
right = 0,0,255
a = 5,1,5
b = 5,1,6
x = 5,1,4
y = 5,1,7
start = 6,1,5
mode = 6,1,4
c = 6,1,0
z = 6,1,1
;
; More examples:
; up = 1,3,0,63|5,2,0       stick up (byte 1 below 64) or hat up
; right = 5,2,2,4           hat right, the hat in the high 4 bits of byte 5
; start = 2,4,0             a button that reads 0 when pressed
;
; Every line must fit in 79 characters.

;
; A short instruction how to dump the values in Linux systems
; 1. lsusb - find a vid/pid of your gamepad
; 2. sudo usbhid-dump -e stream -d vid:pid
; 3. press a button, see which byte changed to which value
; 4. fill the ini file
; 5. profit :)
; Or use the hidmapper: https://karabas.uk/hidmapper/
;

; PS: WARNING! comments are not possible, remove all of them from your ini.
